cmake_minimum_required(VERSION 3.24)
if(APPLE)
  set(CMAKE_OSX_DEPLOYMENT_TARGET "14.4" CACHE STRING "Minimum supported macOS version")
endif()
project(GlassStatsQt VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
set(GLASSSTATS_VERSION "0.2.0-alpha.1")
find_package(Qt6 6.12.0 EXACT REQUIRED COMPONENTS Core Gui Quick QuickControls2 Widgets Network Concurrent Test)
qt_standard_project_setup(REQUIRES 6.12)
add_library(glassstats_core src/core/Core.cpp src/core/Core.h)
target_include_directories(glassstats_core PUBLIC src)
target_link_libraries(glassstats_core PUBLIC Qt6::Core Qt6::Gui)
set(platform_sources src/platform/Platform.h src/platform/Platform.cpp)
if(WIN32)
  list(APPEND platform_sources src/platform/windows/WindowsPlatform.cpp src/platform/windows/WindowsPlatform.h)
elseif(APPLE)
  enable_language(OBJCXX)
  list(APPEND platform_sources src/platform/macos/MacPlatform.mm src/platform/macos/MacPlatform.h)
else()
  find_package(X11)
  list(APPEND platform_sources src/platform/linux/LinuxPlatform.cpp src/platform/linux/LinuxPlatform.h)
endif()
qt_add_executable(GlassStats WIN32 MACOSX_BUNDLE src/main.cpp src/app/Controller.cpp src/app/Controller.h src/app/Updater.cpp src/app/Updater.h src/qa/Validation.cpp src/qa/Validation.h ${platform_sources})
qt_add_resources(GlassStats app_resources PREFIX "/" FILES qml/Main.qml qml/AdaptiveLabel.qml qml/AdaptiveIcon.qml qml/MetricSection.qml qml/SettingsWindow.qml qml/Graph.qml qml/UsageRing.qml qml/DetailPopup.qml assets/GlassStats.ico)
target_link_libraries(GlassStats PRIVATE glassstats_core Qt6::Quick Qt6::QuickControls2 Qt6::Widgets Qt6::Network Qt6::Concurrent)
target_compile_definitions(GlassStats PRIVATE GLASSSTATS_VERSION="${GLASSSTATS_VERSION}")
if(WIN32)
  target_compile_definitions(GlassStats PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
  target_link_libraries(GlassStats PRIVATE pdh dxgi dwmapi shcore advapi32 shell32 user32 gdi32 wtsapi32 psapi)
  target_sources(GlassStats PRIVATE packaging/windows/app.rc)
elseif(APPLE)
  target_link_libraries(GlassStats PRIVATE "-framework AppKit" "-framework ApplicationServices" "-framework IOKit" "-framework Metal" "-framework ScreenCaptureKit" "-framework Carbon")
  set_source_files_properties(src/platform/macos/MacPlatform.mm PROPERTIES COMPILE_OPTIONS "-fobjc-arc")
  set_target_properties(GlassStats PROPERTIES MACOSX_BUNDLE_GUI_IDENTIFIER "io.github.eduardojsborges.GlassStatsQt" MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/packaging/macos/Info.plist.in" MACOSX_BUNDLE_BUNDLE_VERSION "0.2.0" MACOSX_BUNDLE_SHORT_VERSION_STRING "${GLASSSTATS_VERSION}")
elseif(X11_FOUND)
  target_compile_definitions(GlassStats PRIVATE GLASSSTATS_X11)
  target_include_directories(GlassStats PRIVATE ${X11_INCLUDE_DIR})
  target_link_libraries(GlassStats PRIVATE ${X11_LIBRARIES} ${CMAKE_DL_LIBS})
endif()
if(MSVC)
  target_compile_options(GlassStats PRIVATE /W4 /utf-8 /permissive-)
  target_compile_options(glassstats_core PRIVATE /W4 /utf-8 /permissive-)
else()
  target_compile_options(GlassStats PRIVATE -Wall -Wextra -Wpedantic)
endif()
include(CTest)
if(BUILD_TESTING)
  qt_add_executable(core_tests tests/CoreTests.cpp)
  target_link_libraries(core_tests PRIVATE glassstats_core Qt6::Test)
  set_target_properties(core_tests PROPERTIES WIN32_EXECUTABLE FALSE)
  add_test(NAME core COMMAND core_tests)
endif()
install(TARGETS GlassStats BUNDLE DESTINATION . RUNTIME DESTINATION .)
qt_generate_deploy_qml_app_script(TARGET GlassStats OUTPUT_SCRIPT deploy_script NO_UNSUPPORTED_PLATFORM_ERROR)
install(SCRIPT ${deploy_script})
install(FILES LICENSE README.md CHANGELOG.md DESTINATION .)
install(DIRECTORY licenses DESTINATION .)
